#include <stdio.h>

int main()
{
	int num, digit;

	printf("enter an integer:");
	scanf("%d", &num);

	if (num < 0)
		num = -num;

	printf("seprated digits:");

	if (num == 0)

	{
		printf("0");
	}
	
	while (num > 0)

	{

		printf("%d", digit);

		num = num/10;

		if (num > 0)
		
		{
			printf(",");
		}
	}

	printf("\n");
	return 0;
}
