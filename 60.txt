#include <stdio.h>

int main()
{
	int isPrime;

	printf("prime numbers between 1 and 500 are:\n\n");

	for(int num=2; num<=500; num++)

	{
		isPrime = 1;

	for(int i=2; i<=num/2; i++)

	{
		if(num % i == 0)
			
		{
			isPrime = 0;
			break;
		}
	}

	if(isPrime == 1)

	{
		printf("%d", num);
	}
	}
	
	printf("\n");
	
	return 0;
}
