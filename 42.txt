#include <stdio.h>

int main()
{
	int n,i;
	char name[50];

	printf("enter your name:");
	scanf("%s",name);	

	printf("enter how many times to print:");
	scanf("%d",&n);

	printf("\nPrinting name 5 times:\n");
	for (i=1; i<=n; i++)
	{
		printf("%d. %s\n", i, name);
	}
	
	return 0;
}
