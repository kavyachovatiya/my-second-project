#include <stdio.h>

int main()
{
	long long num;
	int count = 0;

	printf("enter an integer:");
	scanf("%lld", &num);

	long long temp = num;

	if (temp == 0)

	{
		count = 1;
	}
	
	else

	{
		if (temp < 0)
		
		{
			temp = -temp;
		}

	while (temp > 0)
		
		{
			temp /= 10;
			count++;
		}
	}

	printf("number of digits in %lld is %d\n", num, count);
	return 0;
}
