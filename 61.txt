#include <stdio.h>

int main()
{
	int isPrime;
	long long sum = 0;

	for(int num=2; num<=500; num++)

	{
		isPrime = 1;

	for(int i=2; i<=num/2; i++)

	{
		if(num % i == 0)
			
		{
			isPrime = 0;
			break;
		}
	}

	if(isPrime == 1)

	{
		sum +=num;
	}
	}
	
	printf("summation of prime numbers between 1 and 500 is: %lld\n", sum);
	
	return 0;
}
