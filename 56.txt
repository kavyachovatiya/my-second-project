#include <stdio.h>

int main()
{
	int num, temp, remainder;
	int sum = 0;

	printf("enter an integer:");
	scanf("%d", &num);

	if (num < 0)
		{
			printf("%d is not a armstrong number.\n", num);
			return 0;
		}
			
	temp = num;

	while (temp > 0)
		
		{
			remainder = temp % 10;
			sum = (remainder * remainder * remainder);
			temp /= 10;
		}

	if (sum == num)
		{
			printf("%d is a armstrong number.\n", num);
		}
	
	else

		{
			printf("%d is not a armstrong number.\n", num);
		}
		
	return 0;
}
