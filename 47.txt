#include <stdio.h>

int main()
{
	int i;
	float num;
	int pos_count=0, neg_count=0, zero_count=0;

	printf("enter 200 numbers:\n");

	for(i=1; i<=200; i++)
	
	{
		printf("value %d:", i);
		scanf("%f", &num);

		if(num > 0)
		{
			pos_count++;
		}
		
		else if (num < 0)

		{
			neg_count++;
		}

		else

		{
			zero_count++;
		}
	}

	printf("\n--- Summary ---\n");
	printf("Positive numbers count: %d\n", pos_count);
	printf("Negative numbers count: %d\n", neg_count);
	printf("zero count: %d\n", zero_count);

	return 0;
}
