#include <stdio.h>

int main()
{
	long long t1 = 1, t2 = 1, nextTerm;
	int n,i;

	printf("enter the number of terms:");
	scanf("%d", &n);

	if(n <= 0)

	{
		printf("please enter a positive integer greater than 0.\n" );
		return 0;
	}
		printf("fibonacci series:");
			
		for (i=1; i<=n; ++i)

		{
			if (i == 1)

			{
				printf("%lld", t1);
			}
	
			if (i == 2)

			{
				printf("%lld", t2);
			}

			nextTerm = t1 + t2;
			t1 = t2;
			t2 = nextTerm;

			printf("%lld", nextTerm);
		}

	printf("\n");
	return 0;
}
