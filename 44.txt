#include <stdio.h>

int main()
{
	int i;
	float number, sum =0.0, mean;

	printf("enter 10 numbers:\n");
	for(i=1; i<=10; i++)

	{
		printf("value %d:", i);
		scanf("%f", &number);
		sum += number;
	}

	mean = sum / 10.0;

	printf("\nSum = %.2f\n", sum);
	printf("Mean = %.2f\n", mean);

	return 0;
}
