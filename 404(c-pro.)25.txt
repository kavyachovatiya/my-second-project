#include<stdio.h>
void main()
{
	float num1, num2;
	
	printf("Enter two numbers:");
	scanf("%f %f", &num1, &num2);

	if (num1 > num2)
{
	printf("Largest: %.2f\n", num1);
	printf("Smallest: %.2f\n", num2);
}
	else if (num2 > num1)
{
	printf("Largest: %.2f\n", num2);
	printf("Smallest: %.2f\n", num1);
}
	else
{
	printf("Both numbers are equal (%.2f).\n", num1);
}
}
