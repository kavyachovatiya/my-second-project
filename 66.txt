#include <stdio.h>

int main()
{
	int i, j;

	printf("--- (A) pattern A ---\n");

	for(i=1, j=5; i<=5; i++, j--)
	
		{
		
			printf("%d %d\n", i, j);
		}



	printf("--- (B) pattern B ---\n");

	for(i=1, j=5; i<=3; i++, j--)
	
		{
		
			printf("%d %d\n", i, j);
			printf("%d %d\n", i, j);
		}

	return 0;
}
