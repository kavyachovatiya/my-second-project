#include <stdio.h>

int main()
{
	int n,i;
	float number, sum =0.0, mean;

	printf("enter the number of values (n):");
	scanf("%d", &n);

	if (n <= 0)

	{
		printf("please enter a positive integer for n.\n");
	}
	
	else
	
	{
	
	printf("enter %d numbers:\n", n);
	for(i=1; i<=n; i++)

	{
		printf("value %d:", i);
		scanf("%f", &number);
		sum += number;
	}

	mean = sum / n;

	printf("\nSum = %.2f\n", sum);
	printf("Mean = %.2f\n", mean);

	}

	return 0;
}
