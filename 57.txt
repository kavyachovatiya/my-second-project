#include <stdio.h>

int main()
{
	int num;

	printf("enter a positive integer:");
	scanf("%d", &num);

	if (num <= 0)
		{
			printf("please enter a positive integer greater than 0.\n");
			return 0;
		}

	printf("factors of %d are:", num);

	for(int i=1; i<=num; i++)

		{
			if (num % i == 0)
				{
					printf("%d", i);
				}
		}

	printf("\n");

	return 0;
}
